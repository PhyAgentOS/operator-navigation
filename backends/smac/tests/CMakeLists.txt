cmake_minimum_required(VERSION 3.16)
project(smac_regression)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_core REQUIRED)
if(PATCHED)
 find_package(navigation_smac_planner REQUIRED)
 set(SMAC navigation_smac_planner)
else()
 find_package(nav2_smac_planner REQUIRED)
 set(SMAC nav2_smac_planner)
endif()
add_executable(probe probe.cpp)
ament_target_dependencies(probe rclcpp nav2_costmap_2d nav2_core ${SMAC})
if(PATCHED)
 target_compile_definitions(probe PRIVATE PATCHED)
endif()
